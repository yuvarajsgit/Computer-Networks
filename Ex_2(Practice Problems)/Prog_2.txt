class Solution {
public:
    bool hasAlternatingBits(int n) 
    {
        if(n == 0)
        {
            return 0;
        }
        string bin = " ";
        while(n>0)
        {
            bin = bin + to_string(n%2);
            n = n/2;
        }
        reverse(bin.begin(),bin.end());
        int count = 0;
        for(int i=0;i<bin.length()-1;i++)
        {
            if(bin[i] != bin[i+1])
            {
                count ++;
            }
        }
        if(count == bin.length()-1)
        {
            return true;
        }
        return false;  
    }
};
