#include <stdio.h>
#include <string.h>
#include <unistd.h>
#include <arpa/inet.h>
#include <sys/socket.h>

int main()
{
    int sockfd;
    char buffer[1024], message[1024];
    struct sockaddr_in serverAddr, clientAddr;
    socklen_t addrLen = sizeof(clientAddr);

    // 1. Create UDP socket
    sockfd = socket(AF_INET, SOCK_DGRAM, 0);
    if (sockfd < 0)
    {
        printf("Socket creation failed\n");
        return 1;
    }

    memset(&serverAddr, 0, sizeof(serverAddr));
    serverAddr.sin_family = AF_INET;
    serverAddr.sin_port = htons(5000);
    serverAddr.sin_addr.s_addr = INADDR_ANY;

    // 2. Bind socket
    if (bind(sockfd, (struct sockaddr *)&serverAddr, sizeof(serverAddr)) < 0)
    {
        printf("Bind failed\n");
        close(sockfd);
        return 1;
    }

    printf("UDP Chat Server is running... Waiting for messages...\n");

    while (1)
    {
        // 3. Receive message from client
        int n = recvfrom(sockfd, buffer, sizeof(buffer) - 1, 0, (struct sockaddr *)&clientAddr, &addrLen);
        if (n < 0)
        {
            printf("Receive failed\n");
            break;
        }
        buffer[n] = '\0';
        
        // If client sends "exit", close chat loop
        if (strcmp(buffer, "exit") == 0)
        {
            printf("Client has disconnected.\n");
            break;
        }

        printf("Client: %s\n", buffer);

        // 4. Send reply to client
        printf("Server reply: ");
        fgets(message, sizeof(message), stdin);
        message[strcspn(message, "\n")] = '\0';

        sendto(sockfd, message, strlen(message), 0, (struct sockaddr *)&clientAddr, addrLen);

        if (strcmp(message, "exit") == 0)
        {
            printf("Chat ended by server.\n");
            break;
        }
    }

    close(sockfd);
    return 0;
}
