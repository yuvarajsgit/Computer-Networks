class Solution {
public:
    int findCheapestPrice(int n, vector<vector<int>>& flights,
                          int src, int dst, int k) {

        const int INF = 1e9;
        vector<int> price(n, INF);

        price[src] = 0;

        // At most k stops = at most k + 1 flights
        for (int i = 0; i <= k; i++) {

            vector<int> temp = price;

            for (auto flight : flights) {
                int from = flight[0];
                int to = flight[1];
                int cost = flight[2];

                if (price[from] != INF) {
                    temp[to] = min(temp[to],
                                   price[from] + cost);
                }
            }

            price = temp;
        }

        return price[dst] == INF ? -1 : price[dst];
    }
};
