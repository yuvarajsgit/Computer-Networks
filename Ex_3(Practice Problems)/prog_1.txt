#include <vector>

class Solution {
public:
    int totalHammingDistance(std::vector<int>& nums) {
        int totalDistance = 0;
        int n = nums.size();
        
        // Loop through all 30 bit positions (enough for numbers up to 10^9)
        for (int i = 0; i < 30; ++i) {
            int countOnes = 0;
            
            // Count how many numbers have the i-th bit set to 1
            for (int num : nums) {
                countOnes += (num >> i) & 1;
            }
            
            // Multiply count of 1s by count of 0s
            totalDistance += countOnes * (n - countOnes);
        }
        
        return totalDistance;
    }
};
