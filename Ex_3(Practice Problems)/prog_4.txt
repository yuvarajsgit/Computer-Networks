class Solution {
public:
    int integerReplacement(int n) {
        long long num = n;
        int operations = 0;
        
        while (num > 1) {
            if ((num & 1) == 0) {
                // If even, divide by 2
                num >>= 1;
            } else if (num == 3 || (num & 3) == 1) {
                // If it ends in 01, or if the number is exactly 3, subtract 1
                num--;
            } else {
                // If it ends in 11, add 1
                num++;
            }
            operations++;
        }
        
        return operations;
    }
};
